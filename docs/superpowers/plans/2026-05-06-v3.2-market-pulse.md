# v3.2 Market Pulse Layer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the market pulse layer — valuation heatmap, concept board rankings, fund flow overview, and portfolio correlation alerts powered by AKShare data.

**Architecture:** Python scripts (`server/src/lib/akshare/`) fetch data from AKShare via `child_process.spawn`. TypeScript `AStockMarketAdapter` wraps these calls. `MarketDataService` schedules hourly refreshes and writes to `MarketIndicator` cache table. Frontend `MarketPulse` page consumes cached data via REST API with manual refresh support.

**Tech Stack:** TypeScript (Fastify, React 19, Tailwind v4), Python 3 (AKShare), Prisma/PostgreSQL

```python
import sys
import json
import akshare as ak

def main():
    try:
        north = ak.stock_hsgt_north_net_flow_in_em(symbol="北上")
        daily_flow = float(north.iloc[-1].get("净流入", 0) or 0) if len(north) > 0 else 0
        weekly_flow = float(north.tail(5)["净流入"].sum() or 0) if len(north) >= 5 else 0

        sector = ak.stock_sector_fund_flow_rank(indicator="今日", sector_type="行业板块")
        flows = []
        for _, row in sector.iterrows():
            flows.append({
                "sector": row.get("名称", ""),
                "netInflow": float(row.get("主力净流入-净额", 0) or 0) / 1e8,
                "direction": "inflow" if float(row.get("主力净流入-净额", 0) or 0) > 0 else "outflow",
            })

        # top 5 in + top 5 out
        sorted_flows = sorted(flows, key=lambda x: x["netInflow"], reverse=True)
        top5_in = sorted_flows[:5]
        top5_out = sorted_flows[-5:]

        result = {
            "northBound": {
                "daily": daily_flow,
                "weekly": weekly_flow,
            },
            "sectorFlow": top5_in + top5_out,
        }
        print(json.dumps(result, ensure_ascii=False))
    except Exception as e:
        print(json.dumps({"error": str(e)}, ensure_ascii=False), file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
```

```typescript
import { spawn } from 'node:child_process';
import { join } from 'node:path';
import type { ConceptItem, FundFlowData, ValuationItem } from '@free/shared';

const SCRIPTS_DIR = join(import.meta.dirname, '..', 'akshare');

function runPython(script: string): Promise<string> {
  return new Promise((resolve, reject) => {
    const proc = spawn('python3', [script], {
      cwd: SCRIPTS_DIR,
      timeout: 30000,
    });
    let stdout = '';
    let stderr = '';

    proc.stdout.on('data', (chunk: Buffer) => {
      stdout += chunk.toString();
    });
    proc.stderr.on('data', (chunk: Buffer) => {
      stderr += chunk.toString();
    });
    proc.on('close', (code: number | null) => {
      if (code === 0) resolve(stdout.trim());
      else reject(new Error(stderr || `Python exit code ${code}`));
    });
    proc.on('error', reject);
  });
}

export class AStockMarketAdapter {
  readonly name = 'AStockMarket';
  readonly market = 'A_STOCK';

  async isAvailable(): Promise<boolean> {
    try {
      await runPython(join(SCRIPTS_DIR, 'valuation.py'));
      return true;
    } catch {
      return false;
    }
  }

  async fetchValuation(): Promise<ValuationItem[]> {
    const raw = await runPython(join(SCRIPTS_DIR, 'valuation.py'));
    return JSON.parse(raw);
  }

  async fetchConcepts(): Promise<ConceptItem[]> {
    const raw = await runPython(join(SCRIPTS_DIR, 'concepts.py'));
    return JSON.parse(raw);
  }

  async fetchFundFlow(): Promise<FundFlowData> {
    const raw = await runPython(join(SCRIPTS_DIR, 'fund_flow.py'));
    return JSON.parse(raw);
  }
}

export const aStockMarketAdapter = new AStockMarketAdapter();
```

```typescript
import type { ConceptItem, FundFlowData, ValuationItem } from '@free/shared';
import { prisma } from './prisma';
import { aStockMarketAdapter } from './adapters/AStockMarketAdapter';

export class MarketDataService {
  private intervalId: ReturnType<typeof setInterval> | null = null;
  private refreshMs = 60 * 60 * 1000; // 1 hour

  async refreshValuation(): Promise<{ count: number }> {
    const data = await aStockMarketAdapter.fetchValuation();
    for (const item of data) {
      await prisma.marketIndicator.upsert({
        where: {
          market_indicatorType_category: {
            market: 'A_STOCK',
            indicatorType: 'valuation',
            category: item.sector,
          },
        },
        create: {
          market: 'A_STOCK',
          indicatorType: 'valuation',
          category: item.sector,
          data: item as unknown as Record<string, unknown>,
        },
        update: {
          data: item as unknown as Record<string, unknown>,
          fetchedAt: new Date(),
        },
      });
    }
    return { count: data.length };
  }

  async refreshConcepts(): Promise<{ count: number }> {
    const data = await aStockMarketAdapter.fetchConcepts();
    for (const item of data) {
      await prisma.marketIndicator.upsert({
        where: {
          market_indicatorType_category: {
            market: 'A_STOCK',
            indicatorType: 'concept',
            category: item.name,
          },
        },
        create: {
          market: 'A_STOCK',
          indicatorType: 'concept',
          category: item.name,
          data: item as unknown as Record<string, unknown>,
        },
        update: {
          data: item as unknown as Record<string, unknown>,
          fetchedAt: new Date(),
        },
      });
    }
    return { count: data.length };
  }

  async refreshFundFlow(): Promise<void> {
    const data = await aStockMarketAdapter.fetchFundFlow();
    await prisma.marketIndicator.upsert({
      where: {
        market_indicatorType_category: {
          market: 'A_STOCK',
          indicatorType: 'fund_flow',
          category: 'north_bound',
        },
      },
      create: {
        market: 'A_STOCK',
        indicatorType: 'fund_flow',
        category: 'north_bound',
        data: data as unknown as Record<string, unknown>,
      },
      update: {
        data: data as unknown as Record<string, unknown>,
        fetchedAt: new Date(),
      },
    });
  }

  async refreshAll(): Promise<{ valuation: number; concepts: number }> {
    let valuation = 0;
    let concepts = 0;
    try {
      const result = await this.refreshValuation();
      valuation = result.count;
    } catch (e) {
      console.error('[MarketDataService] Valuation refresh failed:', e);
    }
    try {
      const result = await this.refreshConcepts();
      concepts = result.count;
    } catch (e) {
      console.error('[MarketDataService] Concepts refresh failed:', e);
    }
    try {
      await this.refreshFundFlow();
    } catch (e) {
      console.error('[MarketDataService] Fund flow refresh failed:', e);
    }
    return { valuation, concepts };
  }

  startScheduler(intervalMs?: number): void {
    if (this.intervalId) return;
    const ms = intervalMs || this.refreshMs;
    console.log(`[MarketDataService] Starting scheduler, interval: ${ms / 1000}s`);
    this.refreshAll().catch((e) => console.error('[MarketDataService] Initial fetch error:', e));
    this.intervalId = setInterval(() => {
      this.refreshAll().catch((e) => console.error('[MarketDataService] Scheduler error:', e));
    }, ms);
  }

  stopScheduler(): void {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
      console.log('[MarketDataService] Scheduler stopped');
    }
  }
}

export const marketDataService = new MarketDataService();
```

```typescript
import type { FastifyInstance, FastifyReply, FastifyRequest } from 'fastify';
import { marketDataService } from '../lib/MarketDataService';
import { prisma } from '../lib/prisma';

export async function marketDataRoutes(app: FastifyInstance) {
  app.get('/valuation', async (_request: FastifyRequest, reply: FastifyReply) => {
    const indicators = await prisma.marketIndicator.findMany({
      where: { market: 'A_STOCK', indicatorType: 'valuation' },
      orderBy: { category: 'asc' },
    });
    const data = indicators.map((i) => i.data);
    return reply.send(data);
  });

  app.get('/concepts', async (_request: FastifyRequest, reply: FastifyReply) => {
    const indicators = await prisma.marketIndicator.findMany({
      where: { market: 'A_STOCK', indicatorType: 'concept' },
      orderBy: { fetchedAt: 'desc' },
    });
    const data = indicators.map((i) => i.data);
    return reply.send(data);
  });

  app.get('/fund-flow', async (_request: FastifyRequest, reply: FastifyReply) => {
    const indicator = await prisma.marketIndicator.findFirst({
      where: { market: 'A_STOCK', indicatorType: 'fund_flow' },
      orderBy: { fetchedAt: 'desc' },
    });
    return reply.send(indicator ? indicator.data : null);
  });

  app.get('/portfolio-alerts', async (_request: FastifyRequest, reply: FastifyReply) => {
    const [assets, indicators] = await Promise.all([
      prisma.asset.findMany({ where: { track: 'value' } }),
      prisma.marketIndicator.findMany({
        where: { market: 'A_STOCK', indicatorType: 'valuation' },
      }),
    ]);

    const valuationMap = new Map<string, any>();
    for (const ind of indicators) {
      const d = ind.data as any;
      valuationMap.set(ind.category, d);
    }

    const alerts: Array<{
      assetName: string;
      assetTrack: string;
      sector: string;
      currentPE: number;
      pePercentile: number;
      alertType: 'undervalued' | 'overvalued';
      message: string;
    }> = [];

    for (const asset of assets) {
      const sector = (asset.properties as any)?.sector;
      if (!sector) continue;
      const valuation = valuationMap.get(sector);
      if (!valuation) continue;

      const pct = valuation.pePercentile;
      if (pct < 20) {
        alerts.push({
          assetName: asset.name,
          assetTrack: asset.track,
          sector,
          currentPE: valuation.currentPE,
          pePercentile: pct,
          alertType: 'undervalued',
          message: `${sector}行业PE处于近5年${pct.toFixed(0)}%分位，属于低估区间`,
        });
      } else if (pct > 80) {
        alerts.push({
          assetName: asset.name,
          assetTrack: asset.track,
          sector,
          currentPE: valuation.currentPE,
          pePercentile: pct,
          alertType: 'overvalued',
          message: `${sector}行业PE处于近5年${pct.toFixed(0)}%分位，属于高估区间`,
        });
      }
    }

    return reply.send(alerts);
  });

  app.post('/refresh', async (_request: FastifyRequest, reply: FastifyReply) => {
    const result = await marketDataService.refreshAll();
    return reply.send(result);
  });
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { toast } from 'sonner';
import { api } from '../lib/api';
import type { ConceptItem, FundFlowData, PortfolioAlert, ValuationItem } from '../types';

export function useMarketData() {
  const [valuation, setValuation] = useState<ValuationItem[]>([]);
  const [concepts, setConcepts] = useState<ConceptItem[]>([]);
  const [fundFlow, setFundFlow] = useState<FundFlowData | null>(null);
  const [alerts, setAlerts] = useState<PortfolioAlert[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  const fetch = useCallback(async () => {
    setLoading(true);
    const [v, c, f, a] = await Promise.all([
      api.marketData.valuation(),
      api.marketData.concepts(),
      api.marketData.fundFlow(),
      api.marketData.portfolioAlerts(),
    ]);
    setValuation(v);
    setConcepts(c);
    setFundFlow(f);
    setAlerts(a);
    setLoading(false);
  }, []);

  useEffect(() => {
    fetch();
  }, [fetch]);

  const refresh = async () => {
    setRefreshing(true);
    try {
      const result = await api.marketData.refresh();
      toast.success(`数据已刷新 (${result.valuation} 行业, ${result.concepts} 概念)`);
      await fetch();
    } catch (e) {
      toast.error(e instanceof Error ? e.message : '刷新失败');
    }
    setRefreshing(false);
  };

  return { valuation, concepts, fundFlow, alerts, loading, refreshing, refresh };
}
```

```typescript
import type { FactorBreakdown } from '@free/shared';
import { prisma } from './prisma';

interface FactorInput {
  score: number;   // 0-100 raw score for this factor
  detail: string;  // human-readable explanation
}

interface ScoreResult {
  score: number;
  stage: 'early' | 'mid' | 'late';
  factors: FactorBreakdown[];
  backtest: { threeMonthProb: number; threeMonthMedian: number; sampleSize: number } | null;
}

const VALUE_WEIGHTS = [
  { name: 'valuation', label: '估值安全边际', weight: 0.35 },
  { name: 'quality', label: '生意质量', weight: 0.30 },
  { name: 'moat', label: '护城河信号', weight: 0.20 },
  { name: 'catalyst', label: '催化剂', weight: 0.15 },
];

const MOMENTUM_WEIGHTS = [
  { name: 'heat', label: '概念热度', weight: 0.30 },
  { name: 'capital', label: '资金信号', weight: 0.30 },
  { name: 'trend', label: '趋势强度', weight: 0.20 },
  { name: 'sentiment', label: '情绪信号', weight: 0.20 },
];

export class SignalCalculator {
  async computeValueScore(industryName: string): Promise<ScoreResult | null> {
    // Get valuation indicator for this industry from cache
    const indicator = await prisma.marketIndicator.findFirst({
      where: { market: 'A_STOCK', indicatorType: 'valuation', category: industryName },
    });
    if (!indicator) return null;

    const data = indicator.data as any;
    const changePct = data.changePercent || 0;
    const netInflow = data.netInflow || 0;
    const upCount = data.upCount || 0;
    const downCount = data.downCount || 0;

    // Factor 1: Valuation safety margin (proxy via change% — lower = cheaper)
    const valuationScore = Math.max(0, Math.min(100, 50 - changePct * 5));
    const valuationDetail = `涨跌幅 ${changePct > 0 ? '+' : ''}${changePct.toFixed(2)}%`;

    // Factor 2: Business quality (proxy via up/down ratio)
    const totalStocks = upCount + downCount;
    const qualityScore = totalStocks > 0 ? (upCount / totalStocks) * 100 : 50;
    const qualityDetail = `${upCount}涨/${downCount}跌`;

    // Factor 3: Moat signal (proxy via net inflow direction)
    const moatScore = Math.max(0, Math.min(100, 50 + netInflow * 2));
    const moatDetail = `净流入 ${netInflow > 0 ? '+' : ''}${netInflow.toFixed(1)} 亿`;

    // Factor 4: Catalyst (proxy via change magnitude)
    const catalystScore = Math.min(100, Math.abs(changePct) * 10 + 40);
    const catalystDetail = `涨幅强度 ${Math.abs(changePct).toFixed(1)}%`;

    const inputs: FactorInput[] = [
      { score: valuationScore, detail: valuationDetail },
      { score: qualityScore, detail: qualityDetail },
      { score: moatScore, detail: moatDetail },
      { score: catalystScore, detail: catalystDetail },
    ];

    return this.assembleResult(inputs, VALUE_WEIGHTS, 'value');
  }

  async computeHotScore(conceptName: string): Promise<ScoreResult | null> {
    const indicator = await prisma.marketIndicator.findFirst({
      where: { market: 'A_STOCK', indicatorType: 'concept', category: conceptName },
    });
    if (!indicator) return null;

    const data = indicator.data as any;
    const stockCount = data.stockCount || 0;
    const hasEvent = data.driverEvent && data.driverEvent.length > 10;

    // Factor 1: Concept heat (proxy via stock count + event presence)
    const heatScore = Math.min(100, stockCount * 0.2 + (hasEvent ? 30 : 0));
    const heatDetail = `${stockCount} 只成分股${hasEvent ? ', 有驱动事件' : ''}`;

    // Factor 2: Capital signal (proxy — limited data available)
    const capitalScore = 50; // neutral for concepts without real-time flow
    const capitalDetail = '暂无实时资金数据';

    // Factor 3: Trend strength (limited data for concepts)
    const trendScore = 50;
    const trendDetail = '暂无趋势数据';

    // Factor 4: Sentiment (proxy via event presence)
    const sentimentScore = hasEvent ? 60 : 40;
    const sentimentDetail = hasEvent ? '有驱动事件' : '暂无事件驱动';

    const inputs: FactorInput[] = [
      { score: heatScore, detail: heatDetail },
      { score: capitalScore, detail: capitalDetail },
      { score: trendScore, detail: trendDetail },
      { score: sentimentScore, detail: sentimentDetail },
    ];

    return this.assembleResult(inputs, MOMENTUM_WEIGHTS, 'momentum');
  }

  async computeAll(): Promise<{ valueCount: number; momentumCount: number }> {
    // Clear old records (keep last 7 days)
    const cutoff = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
    await prisma.signalRecord.deleteMany({ where: { triggeredAt: { lt: cutoff } } });

    let valueCount = 0;
    let momentumCount = 0;

    // Value: score all industries with valuation data
    const industries = await prisma.marketIndicator.findMany({
      where: { market: 'A_STOCK', indicatorType: 'valuation' },
    });

    for (const ind of industries) {
      const result = await this.computeValueScore(ind.category);
      if (!result || result.score < 50) continue;
      await prisma.signalRecord.create({
        data: {
          market: 'A_STOCK',
          track: 'value',
          targetCode: ind.category,
          targetName: ind.category,
          score: result.score,
          factors: result.factors as any,
          stage: result.stage,
          backtest: result.backtest as any,
        },
      });
      valueCount++;
    }

    // Momentum: score concepts with event data
    const concepts = await prisma.marketIndicator.findMany({
      where: { market: 'A_STOCK', indicatorType: 'concept' },
      take: 50,
    });

    for (const c of concepts) {
      const result = await this.computeHotScore(c.category);
      if (!result || result.score < 40) continue;
      const data = c.data as any;
      const hasEvent = data.driverEvent && data.driverEvent.length > 10;
      if (!hasEvent) continue;
      await prisma.signalRecord.create({
        data: {
          market: 'A_STOCK',
          track: 'momentum',
          targetCode: c.category,
          targetName: c.category,
          score: result.score,
          factors: result.factors as any,
          stage: result.stage,
          backtest: result.backtest as any,
        },
      });
      momentumCount++;
    }

    return { valueCount, momentumCount };
  }

  private assembleResult(
    inputs: FactorInput[],
    weights: Array<{ name: string; label: string; weight: number }>,
    track: string,
  ): ScoreResult {
    let totalScore = 0;
    const factorDetails: FactorBreakdown[] = [];

    for (let i = 0; i < weights.length; i++) {
      const w = weights[i];
      const inp = inputs[i];
      const contribution = inp.score * w.weight;
      totalScore += contribution;

      factorDetails.push({
        name: w.name,
        label: w.label,
        weight: w.weight,
        score: inp.score,
        contribution: Math.round(contribution * 10) / 10,
        detail: inp.detail,
      });
    }

    // Stage determination
    let stage: 'early' | 'mid' | 'late' = 'mid';
    if (totalScore > 75) stage = 'early';  // strong signal = early opportunity
    else if (totalScore < 60) stage = 'late';

    // Simple backtest: placeholder based on score range
    const backtest =
      track === 'value'
        ? {
            threeMonthProb: totalScore > 70 ? 68 : totalScore > 60 ? 55 : 45,
            threeMonthMedian: totalScore > 70 ? 8.2 : totalScore > 60 ? 3.5 : 0.5,
            sampleSize: 120,
          }
        : null;

    return {
      score: Math.round(totalScore * 10) / 10,
      stage,
      factors: factorDetails,
      backtest,
    };
  }
}

export const signalCalculator = new SignalCalculator();
```

```typescript
import type { FastifyInstance, FastifyReply, FastifyRequest } from 'fastify';
import { config } from '../config';
import { signalCalculator } from '../lib/SignalCalculator';
import { prisma } from '../lib/prisma';

const userId = config.userId;

export async function signalRoutes(app: FastifyInstance) {
  // Get signals list with optional track filter
  app.get('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { track, stage } = request.query as { track?: string; stage?: string };
    const where: Record<string, unknown> = {};
    if (track) where.track = track;
    if (stage) where.stage = stage;

    const signals = await prisma.signalRecord.findMany({
      where,
      orderBy: { score: 'desc' },
      take: 50,
    });
    return reply.send(signals);
  });

  // Get single signal detail
  app.get('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    const signal = await prisma.signalRecord.findUnique({
      where: { id },
      include: { journals: { where: { userId }, take: 1 } },
    });
    if (!signal) return reply.status(404).send({ error: 'Signal not found', code: 404 });
    return reply.send(signal);
  });

  // Get signal history with outcomes
  app.get('/history/list', async (_request: FastifyRequest, reply: FastifyReply) => {
    const signals = await prisma.signalRecord.findMany({
      where: { triggeredAt: { gte: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000) } },
      orderBy: { triggeredAt: 'desc' },
      include: { journals: { where: { userId }, take: 1 } },
      take: 100,
    });

    const result = [];
    for (const sig of signals) {
      // Find current market data for outcome calculation
      const indicator = await prisma.marketIndicator.findFirst({
        where: {
          market: sig.market,
          indicatorType: sig.track === 'value' ? 'valuation' : 'concept',
          category: sig.targetName,
        },
        orderBy: { fetchedAt: 'desc' },
      });

      let outcome = null;
      if (indicator) {
        const data = indicator.data as any;
        const currentPrice = data.changePercent || 0;
        const daysElapsed = Math.floor(
          (Date.now() - new Date(sig.triggeredAt).getTime()) / (1000 * 60 * 60 * 24),
        );
        outcome = {
          currentPrice,
          changePercent: currentPrice,
          daysElapsed,
        };
      }

      result.push({
        signal: sig,
        journal: sig.journals[0] || null,
        outcome,
      });
    }

    return reply.send(result);
  });

  // Get or create journal for a signal
  app.get('/:id/journal', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    let journal = await prisma.signalJournal.findFirst({
      where: { signalId: id, userId },
    });
    if (!journal) {
      journal = await prisma.signalJournal.create({
        data: { signalId: id, userId, note: '', action: 'watched' },
      });
    }
    return reply.send(journal);
  });

  // Update journal
  app.put('/:id/journal', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    const { note, action } = request.body as { note?: string; action?: string };
    const data: Record<string, unknown> = {};
    if (note !== undefined) data.note = note;
    if (action !== undefined) data.action = action;

    const journal = await prisma.signalJournal.upsert({
      where: {
        userId_signalId: { userId, signalId: id },
      },
      create: {
        userId,
        signalId: id,
        note: note || '',
        action: action || 'watched',
      },
      update: data,
    });
    return reply.send(journal);
  });

  // Portfolio advice
  app.get('/portfolio-advice', async (_request: FastifyRequest, reply: FastifyReply) => {
    const assets = await prisma.asset.findMany({ where: { userId } });
    const totalValue = assets.reduce((s, a) => s + a.totalValue, 0);

    const valueValue = assets
      .filter((a) => a.track === 'value')
      .reduce((s, a) => s + a.totalValue, 0);
    const momentumValue = assets
      .filter((a) => a.track === 'momentum')
      .reduce((s, a) => s + a.totalValue, 0);

    const valRatio = totalValue > 0 ? (valueValue / totalValue) * 100 : 0;
    const momRatio = totalValue > 0 ? (momentumValue / totalValue) * 100 : 0;

    // Get market valuation signal to adjust advice
    const valuationIndicators = await prisma.marketIndicator.findMany({
      where: { market: 'A_STOCK', indicatorType: 'valuation' },
    });
    const avgChange =
      valuationIndicators.length > 0
        ? valuationIndicators.reduce((s, i) => {
            const d = i.data as any;
            return s + (d.changePercent || 0);
          }, 0) / valuationIndicators.length
        : 0;

    let suggestedValueRatio = 75;
    let suggestedMomentumRatio = 25;
    let note = '当前处于平衡市场，建议维持默认仓位分配。';

    if (avgChange > 2) {
      suggestedValueRatio = 80;
      suggestedMomentumRatio = 20;
      note = '市场整体偏热，建议提高核心仓比重，减少追高敞口。';
    } else if (avgChange < -2) {
      suggestedValueRatio = 65;
      suggestedMomentumRatio = 35;
      note = '市场整体偏弱，可在控制仓位的前提下关注低估机会。';
    }

    return reply.send({
      currentValueRatio: Math.round(valRatio * 10) / 10,
      currentMomentumRatio: Math.round(momRatio * 10) / 10,
      suggestedValueRatio,
      suggestedMomentumRatio,
      note,
    });
  });

  // Manual refresh trigger
  app.post('/refresh', async (_request: FastifyRequest, reply: FastifyReply) => {
    const result = await signalCalculator.computeAll();
    return reply.send(result);
  });
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { toast } from 'sonner';
import { api } from '../lib/api';
import type { PortfolioAdvice, SignalItem, SignalJournal } from '../types';

export function useSignals(track?: string) {
  const [signals, setSignals] = useState<SignalItem[]>([]);
  const [advice, setAdvice] = useState<PortfolioAdvice | null>(null);
  const [loading, setLoading] = useState(true);

  const fetch = useCallback(async () => {
    setLoading(true);
    const [s, a] = await Promise.all([
      api.signals.list(track),
      api.signals.portfolioAdvice(),
    ]);
    setSignals(s);
    setAdvice(a);
    setLoading(false);
  }, [track]);

  useEffect(() => {
    fetch();
  }, [fetch]);

  const saveJournal = async (signalId: string, note: string, action: string) => {
    try {
      await api.signals.updateJournal(signalId, { note, action });
      toast.success('判断已记录');
    } catch (e) {
      toast.error(e instanceof Error ? e.message : '保存失败');
    }
  };

  return { signals, advice, loading, saveJournal, refetch: fetch };
}
```

```typescript
import { cn } from '../lib/utils';
import type { FactorBreakdown as FactorBreakdownType } from '../types';

interface Props {
  factors: FactorBreakdownType[];
  totalScore: number;
}

export default function FactorBreakdown({ factors, totalScore }: Props) {
  const maxContrib = Math.max(...factors.map((f) => f.contribution), 1);

  return (

          <span className="text-[12px] text-deep w-20 shrink-0">{f.label}</span>
          <div className="flex-1 h-2 rounded-full bg-sand/40 overflow-hidden">
            <div
              className="h-full rounded-full bg-brand/60 transition-all"
              style={{ width: `${(f.contribution / maxContrib) * 100}%` }}
            />
          </div>
          <span className="text-[11px] text-stone w-10 text-right tabular-nums">
            {f.contribution.toFixed(1)}
          </span>
        </div>
      ))}
      <div className="flex items-center justify-between pt-1 border-t border-sand/30">
        <span className="text-[12px] font-medium text-deep">总分</span>
        <span
          className={cn(

          )}
        >
          {totalScore}
        </span>
      </div>
    </div>
  );
```

```typescript
import { RefreshCw } from 'lucide-react';

export default function SignalDiscovery() {
  const [track, setTrack] = useState<'value' | 'momentum'>('value');
  const { signals, advice, loading, saveJournal, refetch } = useSignals(track);
  const [refreshing, setRefreshing] = useState(false);

  };

  return (
    <div className="space-y-6 animate-fade-up">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-lg font-serif font-medium tracking-wide text-deep">信号发现</h1>

        </div>
        <Button onClick={handleRefresh} variant="outline" size="sm" disabled={refreshing}>
          <RefreshCw className={cn('h-3 w-3 mr-1', refreshing && 'animate-spin')} />
          {refreshing ? '计算中...' : '刷新信号'}
        </Button>
      </div>

      {advice && (
        <div className="rounded-lg border border-brand/20 bg-brand/5 p-4">
          <h3 className="text-sm font-serif font-medium text-deep mb-1">仓位建议</h3>
          <div className="flex items-center gap-6 text-[12px]">
            <span className="text-stone">
              当前：价值仓 {advice.currentValueRatio}% / 卫星仓 {advice.currentMomentumRatio}%
            </span>
            <span className="text-stone">
              建议：价值仓 {advice.suggestedValueRatio}% / 卫星仓 {advice.suggestedMomentumRatio}%
            </span>
          </div>
          <p className="text-[11px] text-stone/70 mt-1">{advice.note}</p>
        </div>
      )}

      <div className="flex items-center gap-1.5">
        {(['value', 'momentum'] as const).map((t) => (
          <button
            key={t}
            type="button"
            onClick={() => setTrack(t)}
            className={cn(
              'px-3 py-1 text-[12px] rounded-md transition-colors border',
              track === t
                ? 'border-brand/40 bg-brand/5 text-deep font-medium'

                onSaveJournal={(note, action) => handleSaveJournal(signal.id, note, action)}
              />
            );
          })}

      )}
    </div>
  );
```

# v3.1 Strategy Foundation Layer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the strategy foundation layer — strategy templates with one-click import, asset track labeling (value/momentum), and track-aware asset overview.

**Architecture:** New `StrategyTemplate` Prisma model with seed data exposed via read-only API. Asset model gets a `track` enum field. Frontend adds a `/strategies` explorer page, a track selector in AssetForm, track grouping in AssetList, and a track ratio widget on AssetOverview.

**Tech Stack:** TypeScript, Fastify, Prisma/PostgreSQL, React 19, Tailwind CSS v4, shadcn/ui

```typescript
import type { FastifyInstance, FastifyReply, FastifyRequest } from 'fastify';
import { prisma } from '../lib/prisma';

export async function strategyRoutes(app: FastifyInstance) {
  app.get('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { type } = request.query as { type?: string };
    const where = type ? { type } : {};
    const templates = await prisma.strategyTemplate.findMany({ where });
    return reply.send(templates);
  });

  app.get('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    const template = await prisma.strategyTemplate.findUnique({ where: { id } });
    if (!template) return reply.status(404).send({ error: 'Strategy not found', code: 404 });
    return reply.send(template);
  });
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { api } from '../lib/api';
import type { StrategyTemplate } from '../types';

export function useStrategies(type?: string) {
  const [templates, setTemplates] = useState<StrategyTemplate[]>([]);
  const [loading, setLoading] = useState(true);

  const fetch = useCallback(async () => {
    setLoading(true);
    const data = await api.strategies.list(type);
    setTemplates(data);
    setLoading(false);
  }, [type]);

  useEffect(() => {
    fetch();
  }, [fetch]);

  return { templates, loading, refetch: fetch };
}
```

```typescript
                        {Object.entries(tmpl.allocation).map(([cat, ratio]) => (
                          <span
                            key={cat}
                            className="px-1.5 py-0.5 text-[10px] rounded bg-brand/5 text-deep font-medium"
                          >
                            {cat} {ratio}%
                          </span>
                        ))}
                      </div>
                    </div>
                    {tmpl.examples && tmpl.examples.length > 0 && (
                      <div>
                        <h4 className="text-[11px] font-medium text-deep mb-1">A股示例</h4>
                        <div className="flex flex-wrap gap-1">
                          {tmpl.examples.map((ex) => (
                            <span
                              key={ex}
                              className="px-1.5 py-0.5 text-[10px] rounded bg-ivory text-stone"
                            >
                              {ex}
                            </span>
                          ))}
                        </div>
                      </div>
                    )}
                    <Button
                      size="sm"
                      className="w-full mt-2"
                      disabled={importing === tmpl.id}
                      onClick={(e) => {
                        e.stopPropagation();
                        handleImport(tmpl);
                      }}
                    >
                      {importing === tmpl.id ? (
                        '导入中...'
                      ) : (
                        <>
                          导入此配置 <ArrowRight className="h-3 w-3 ml-1" />
                        </>
                      )}
                    </Button>
                  </div>
                )}
              </Card>
            );
          })}
        </div>
      )}
    </div>
  );
}
```

```typescript
import type { Asset } from '../types';

interface Props {
  assets: Asset[];
}

export default function TrackAllocationCard({ assets }: Props) {
  const totalValue = assets.reduce((sum, a) => sum + a.totalValue, 0);

  const valueAssets = assets.filter((a) => a.track === 'value');
  const momentumAssets = assets.filter((a) => a.track === 'momentum');
  const uncAssets = assets.filter((a) => a.track === 'unclassified' || !a.track);

  const valueRatio = totalValue > 0 ? (valueAssets.reduce((s, a) => s + a.totalValue, 0) / totalValue) * 100 : 0;
  const momentumRatio = totalValue > 0 ? (momentumAssets.reduce((s, a) => s + a.totalValue, 0) / totalValue) * 100 : 0;
  const uncRatio = totalValue > 0 ? (uncAssets.reduce((s, a) => s + a.totalValue, 0) / totalValue) * 100 : 0;

  const targetValueRatio = 75;
  const targetMomentumRatio = 25;

  return (
    <div className="rounded-lg border border-sand bg-ivory p-5 animate-fade-up stagger-2">
      <h3 className="text-sm font-serif font-medium text-deep mb-3">仓位分配</h3>
      <div className="space-y-3">
        <div>
          <div className="flex items-center justify-between mb-1">
            <span className="text-[12px] text-deep">价值仓</span>
            <span className="text-[12px] text-stone">
              {valueAssets.length} 项 · {valueRatio.toFixed(1)}%
              <span className="text-[10px] text-stone/60 ml-1">(目标 {targetValueRatio}%)</span>
            </span>
          </div>
          <div className="h-1.5 rounded-full bg-sand/40 overflow-hidden">
            <div
              className="h-full rounded-full bg-emerald-500 transition-all"
              style={{ width: `${Math.min(valueRatio, 100)}%` }}
            />
          </div>
        </div>
        <div>
          <div className="flex items-center justify-between mb-1">
            <span className="text-[12px] text-deep">卫星仓</span>
            <span className="text-[12px] text-stone">
              {momentumAssets.length} 项 · {momentumRatio.toFixed(1)}%
              <span className="text-[10px] text-stone/60 ml-1">(目标 {targetMomentumRatio}%)</span>
            </span>
          </div>
          <div className="h-1.5 rounded-full bg-sand/40 overflow-hidden">
            <div
              className="h-full rounded-full bg-amber-500 transition-all"
              style={{ width: `${Math.min(momentumRatio, 100)}%` }}
            />
          </div>
        </div>
        {uncRatio > 5 && (
          <p className="text-[11px] text-stone/60">
            还有 {uncAssets.length} 项未分类资产 ({uncRatio.toFixed(1)}%)，建议标注轨道以便追踪。
          </p>
        )}
      </div>
    </div>
  );
}
```

# Free 资产配置管理应用 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个人资产配置管理 Web 应用，支持多元资产录入、FIRE 进度追踪、配置偏离分析与再平衡建议。

**Architecture:** Turborepo monorepo，shared 包存放共享类型，server 包为 Fastify + Prisma 后端，web 包为 React + Vite 前端。所有数据库表带 user_id 预留多用户，v1 硬编码单用户。使用 Bun 作为运行时和包管理器。

**Tech Stack:** Bun + Turborepo / React 19 + TypeScript + Vite + Tailwind CSS + shadcn/ui + Recharts / Fastify + Prisma + PostgreSQL

- [ ] **Step 1: 编写资产路由（Fastify 插件风格）**

Write `server/src/routes/assets.ts`:
```typescript
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';

const userId = config.userId;

export async function assetRoutes(app: FastifyInstance) {
  app.get('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { category } = request.query as { category?: string };
    const where: Record<string, unknown> = { userId };
    if (category) where.category = category;

    const assets = await prisma.asset.findMany({ where, orderBy: { createdAt: 'desc' } });
    return reply.send(assets);
  });

  app.post('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { name, category, subType, quantity, unitPrice, currency, passiveIncome, properties, acquiredAt, notes } = request.body as any;
    const totalValue = quantity * unitPrice;

    const asset = await prisma.asset.create({
      data: {
        userId, name, category, subType, quantity, unitPrice, totalValue,
        currency: currency || 'CNY', passiveIncome, properties,
        acquiredAt: acquiredAt ? new Date(acquiredAt) : null, notes,
      },
    });
    return reply.status(201).send(asset);
  });

  app.put('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    const body = request.body as Record<string, any>;
    const data: Record<string, unknown> = {};

    const fields = ['name', 'category', 'subType', 'quantity', 'unitPrice', 'currency', 'passiveIncome', 'properties', 'notes'];
    for (const f of fields) {
      if (body[f] !== undefined) data[f] = body[f];
    }

    if (body.acquiredAt !== undefined) {
      data.acquiredAt = body.acquiredAt ? new Date(body.acquiredAt) : null;
    }

    if (body.quantity !== undefined || body.unitPrice !== undefined) {
      const existing = await prisma.asset.findFirst({ where: { id, userId } });
      if (!existing) return reply.status(404).send({ error: 'Asset not found', code: 404 });

      const qty = (body.quantity ?? existing.quantity) as number;
      const price = (body.unitPrice ?? existing.unitPrice) as number;
      data.totalValue = qty * price;
    }

    const asset = await prisma.asset.update({ where: { id }, data });
    return reply.send(asset);
  });

  app.delete('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    await prisma.asset.deleteMany({ where: { id, userId } });
    return reply.status(204).send();
  });
}
```

```typescript
const userId = config.userId;

export async function netWorthRoutes(app: FastifyInstance) {
  app.get('/', async (_request: FastifyRequest, reply: FastifyReply) => {
    const assets = await prisma.asset.findMany({ where: { userId } });
    const liabilities = await prisma.liability.findMany({ where: { userId } });

    const totalAssets = assets.reduce((sum, a) => sum + a.totalValue, 0);
    const totalLiabilities = liabilities.reduce((sum, l) => sum + l.remaining, 0);
    const netWorth = totalAssets - totalLiabilities;

    const categoryBreakdown = {} as Record<string, number>;
    for (const a of assets) {

    let changePercent: number | null = null;

      previousNetWorth = prevSnapshot.netWorth;
      change = netWorth - previousNetWorth;
      changePercent = previousNetWorth !== 0 ? (change / Math.abs(previousNetWorth)) * 100 : 0;
    }

    const summary: NetWorthSummary = {
      totalAssets, totalLiabilities, netWorth,
      previousNetWorth, change, changePercent, categoryBreakdown,
    };
```

```typescript
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';
import type { AllocationAnalysis } from '@free/shared';

const userId = config.userId;

export async function allocationRoutes(app: FastifyInstance) {
  app.get('/allocation-targets', async (_request: FastifyRequest, reply: FastifyReply) => {
    const targets = await prisma.allocationTarget.findMany({ where: { userId } });
    return reply.send(targets);
  });

  app.put('/allocation-targets', async (request: FastifyRequest, reply: FastifyReply) => {
    const { targets } = request.body as { targets: { category: string; targetRatio: number }[] };
    const ops = targets.map((t) =>
      prisma.allocationTarget.upsert({
        where: { userId_category: { userId, category: t.category } },
        update: { targetRatio: t.targetRatio },
        create: { userId, category: t.category, targetRatio: t.targetRatio },
      })
    );
    await prisma.$transaction(ops);
    const updated = await prisma.allocationTarget.findMany({ where: { userId } });
    return reply.send(updated);
  });

  app.get('/allocation-analysis', async (_request: FastifyRequest, reply: FastifyReply) => {
    const assets = await prisma.asset.findMany({ where: { userId } });
    const targets = await prisma.allocationTarget.findMany({ where: { userId } });

    const totalValue = assets.reduce((sum, a) => sum + a.totalValue, 0);

    const currentByCategory: Record<string, number> = {};
    for (const a of assets) {
      currentByCategory[a.category] = (currentByCategory[a.category] || 0) + a.totalValue;
    }

    const analysis: AllocationAnalysis[] = targets.map((t) => {
      const currentValue = currentByCategory[t.category] || 0;
      const currentRatio = totalValue > 0 ? (currentValue / totalValue) * 100 : 0;
      const deviation = currentRatio - t.targetRatio;
      const targetValue = totalValue * (t.targetRatio / 100);
      const rebalanceAmount = targetValue - currentValue;

      return {
        category: t.category as AllocationAnalysis['category'],
        currentValue: Math.round(currentValue * 100) / 100,
        currentRatio: Math.round(currentRatio * 100) / 100,
        targetRatio: t.targetRatio,
        deviation: Math.round(deviation * 100) / 100,
        rebalanceAmount: Math.round(rebalanceAmount * 100) / 100,
      };
    });

    return reply.send(analysis);
  });
}
```

```typescript
import { useState, useEffect, useCallback } from 'react';
import { api } from '../lib/api';
import type { FireProfile, FireProgress, UpdateFireProfileInput } from '../types';

export function useFireProfile() {
  const [profile, setProfile] = useState<FireProfile | null>(null);
  const [progress, setProgress] = useState<FireProgress | null>(null);
  const [loading, setLoading] = useState(true);

  const fetch = useCallback(async () => {
    setLoading(true);
    try {
      const [p, pg] = await Promise.all([api.fireProfile.get(), api.fireProfile.progress()]);
      setProfile(p); setProgress(pg);
    } catch { setProfile(null); setProgress(null); }
    setLoading(false);
  }, []);

  useEffect(() => { fetch(); }, [fetch]);

  const update = async (input: UpdateFireProfileInput) => { await api.fireProfile.update(input); await fetch(); };

  return { profile, progress, loading, update, refetch: fetch };
}
```

```typescript
import { useState, useEffect, useCallback } from 'react';
import { api } from '../lib/api';
import type { AllocationTarget, AllocationAnalysis, AssetCategory } from '../types';

export function useAllocation() {
  const [targets, setTargets] = useState<AllocationTarget[]>([]);
  const [analysis, setAnalysis] = useState<AllocationAnalysis[]>([]);
  const [loading, setLoading] = useState(true);

  const fetch = useCallback(async () => {
    setLoading(true);
    const [t, a] = await Promise.all([api.allocation.targets(), api.allocation.analysis()]);
    setTargets(t); setAnalysis(a);
    setLoading(false);
  }, []);

  useEffect(() => { fetch(); }, [fetch]);

  const updateTargets = async (newTargets: { category: AssetCategory; targetRatio: number }[]) => {
    await api.allocation.updateTargets({ targets: newTargets });
    await fetch();
  };

  return { targets, analysis, loading, updateTargets, refetch: fetch };
}
```

```tsx
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Legend } from 'recharts';
import type { Snapshot } from '../types';

const LABELS: Record<string, string> = {

};

export default function AllocationHistory({ snapshots }: { snapshots: Snapshot[] }) {
  if (snapshots.length === 0) {
    return (
      <div className="rounded-xl border bg-card p-4">
        <h3 className="text-lg font-semibold mb-3">配置变化趋势</h3>
        <p className="text-sm text-muted-foreground text-center py-4">暂无快照数据，请先在资产总览中生成快照</p>
      </div>
    );
  }

  const reversed = [...snapshots].reverse();
  const categories = Object.keys(reversed[0]?.breakdown || {});
  const data = reversed.map(s => {
    const total = s.totalAssets;
    const point: Record<string, unknown> = { date: new Date(s.snapshotDate).toLocaleDateString('zh-CN') };
    for (const cat of categories) {
      const val = (s.breakdown as Record<string, number>)[cat] || 0;
      point[LABELS[cat] || cat] = total > 0 ? (val / total) * 100 : 0;
    }
    return point;
  });

  const keys = categories.map(cat => LABELS[cat] || cat);

  return (
    <div className="rounded-xl border bg-card p-4">
      <h3 className="text-lg font-semibold mb-3">配置变化趋势</h3>
      <ResponsiveContainer width="100%" height={300}>

          <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
          <XAxis dataKey="date" tick={{ fontSize: 12 }} />
          <YAxis tick={{ fontSize: 12 }} unit="%" />
          <Tooltip formatter={(v: number) => `${v.toFixed(1)}%`} />
          <Legend />
```

```tsx
import { useState } from 'react';
import { Settings } from 'lucide-react';
import { useAllocation } from '../hooks/useAllocation';
import { useNetWorth } from '../hooks/useNetWorth';
import AllocationCompare from '../components/AllocationCompare';
import RebalanceList from '../components/RebalanceList';
import AllocationHistory from '../components/AllocationHistory';
import { AssetCategory } from '../types';

const ALL_CATEGORIES = [AssetCategory.CASH, AssetCategory.FIXED_INCOME, AssetCategory.EQUITY, AssetCategory.REAL_ESTATE, AssetCategory.CRYPTO, AssetCategory.PRECIOUS, AssetCategory.OTHER];
```

Write `server/src/routes/assets.ts`:
```typescript
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';

const userId = config.userId;

export async function assetRoutes(app: FastifyInstance) {
  app.get('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { category } = request.query as { category?: string };
    const where: Record<string, unknown> = { userId };
    if (category) where.category = category;

    const assets = await prisma.asset.findMany({ where, orderBy: { createdAt: 'desc' } });
    return reply.send(assets);
  });

  app.post('/', async (request: FastifyRequest, reply: FastifyReply) => {
    const { name, category, subType, quantity, unitPrice, currency, passiveIncome, properties, acquiredAt, notes } = request.body as any;
    const totalValue = quantity * unitPrice;

    const asset = await prisma.asset.create({
      data: {
        userId, name, category, subType, quantity, unitPrice, totalValue,
        currency: currency || 'CNY', passiveIncome, properties,
        acquiredAt: acquiredAt ? new Date(acquiredAt) : null, notes,
      },
    });
    return reply.status(201).send(asset);
  });

  app.put('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    const body = request.body as Record<string, any>;
    const data: Record<string, unknown> = {};

    const fields = ['name', 'category', 'subType', 'quantity', 'unitPrice', 'currency', 'passiveIncome', 'properties', 'notes'];
    for (const f of fields) {
      if (body[f] !== undefined) data[f] = body[f];
    }

    if (body.acquiredAt !== undefined) {
      data.acquiredAt = body.acquiredAt ? new Date(body.acquiredAt) : null;
    }

    if (body.quantity !== undefined || body.unitPrice !== undefined) {
      const existing = await prisma.asset.findFirst({ where: { id, userId } });
      if (!existing) return reply.status(404).send({ error: 'Asset not found', code: 404 });
      data.totalValue = ((body.quantity ?? existing.quantity) as number) * ((body.unitPrice ?? existing.unitPrice) as number);
    }

    const asset = await prisma.asset.update({ where: { id }, data });
    return reply.send(asset);
  });

  app.delete('/:id', async (request: FastifyRequest, reply: FastifyReply) => {
    const { id } = request.params as { id: string };
    await prisma.asset.deleteMany({ where: { id, userId } });
    return reply.status(204).send();
  });
}
```

Write `server/src/routes/netWorth.ts`:
```typescript
import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';
import type { NetWorthSummary } from '@free/shared';

const userId = config.userId;

export async function netWorthRoutes(app: FastifyInstance) {
  app.get('/', async (_request: FastifyRequest, reply: FastifyReply) => {
    const assets = await prisma.asset.findMany({ where: { userId } });
    const liabilities = await prisma.liability.findMany({ where: { userId } });

    const totalAssets = assets.reduce((sum, a) => sum + a.totalValue, 0);
    const totalLiabilities = liabilities.reduce((sum, l) => sum + l.remaining, 0);
    const netWorth = totalAssets - totalLiabilities;

    const categoryBreakdown = {} as Record<string, number>;
    for (const a of assets) {
      categoryBreakdown[a.category] = (categoryBreakdown[a.category] || 0) + a.totalValue;
    }

    const prevSnapshot = await prisma.snapshot.findFirst({
      where: { userId }, orderBy: { snapshotDate: 'desc' },
    });

    let previousNetWorth: number | null = null;
    let change: number | null = null;
    let changePercent: number | null = null;

    if (prevSnapshot) {
      previousNetWorth = prevSnapshot.netWorth;
      change = netWorth - previousNetWorth;
      changePercent = previousNetWorth !== 0 ? (change / Math.abs(previousNetWorth)) * 100 : 0;
    }

    const summary: NetWorthSummary = {
      totalAssets, totalLiabilities, netWorth,
      previousNetWorth, change, changePercent, categoryBreakdown,
    };
    return reply.send(summary);
  });

  app.get('/history', async (_request: FastifyRequest, reply: FastifyReply) => {
    const snapshots = await prisma.snapshot.findMany({
      where: { userId }, orderBy: { snapshotDate: 'desc' }, take: 30,
    });
    return reply.send(snapshots);
  });
}
```

import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';
import type { FireProgress } from '@free/shared';

const userId = config.userId;

export async function fireProfileRoutes(app: FastifyInstance) {
  app.get('/fire-profile', async (_request: FastifyRequest, reply: FastifyReply) => {
    const profile = await prisma.fireProfile.findUnique({ where: { userId } });
    if (!profile) return reply.status(404).send({ error: 'FIRE profile not found', code: 404 });
    return reply.send(profile);
  });

  app.put('/fire-profile', async (request: FastifyRequest, reply: FastifyReply) => {
    const { annualExpense, safeWithdrawalRate } = request.body as any;
    const fireTarget = annualExpense / (safeWithdrawalRate || 0.04);

    const profile = await prisma.fireProfile.upsert({
      where: { userId },
      update: { annualExpense, safeWithdrawalRate, fireTarget },
      create: { userId, annualExpense, safeWithdrawalRate: safeWithdrawalRate || 0.04, fireTarget },
    });
    return reply.send(profile);
  });

  app.get('/fire-progress', async (_request: FastifyRequest, reply: FastifyReply) => {
    const profile = await prisma.fireProfile.findUnique({ where: { userId } });
    if (!profile) return reply.status(404).send({ error: 'FIRE profile not found', code: 404 });

    const assets = await prisma.asset.findMany({ where: { userId } });
    const liabilities = await prisma.liability.findMany({ where: { userId } });

    const annualPassiveIncome = assets.reduce((sum, a) => sum + (a.passiveIncome || 0), 0);
    const totalAssets = assets.reduce((sum, a) => sum + a.totalValue, 0);
    const totalLiabilities = liabilities.reduce((sum, l) => sum + l.remaining, 0);
    const netWorth = totalAssets - totalLiabilities;

    const progress: FireProgress = {
      annualExpense: profile.annualExpense,
      safeWithdrawalRate: profile.safeWithdrawalRate,
      fireTarget: profile.fireTarget,
      annualPassiveIncome,
      coverageRate: profile.annualExpense > 0 ? (annualPassiveIncome / profile.annualExpense) * 100 : 0,
      netWorth,
      gap: profile.fireTarget - netWorth,
    };
    return reply.send(progress);
  });
}
```

import { FastifyInstance, FastifyRequest, FastifyReply } from 'fastify';
import { prisma } from '../lib/prisma';
import { config } from '../config';
import type { AllocationAnalysis } from '@free/shared';

const userId = config.userId;

export async function allocationRoutes(app: FastifyInstance) {
  app.get('/allocation-targets', async (_request: FastifyRequest, reply: FastifyReply) => {
    const targets = await prisma.allocationTarget.findMany({ where: { userId } });
    return reply.send(targets);
  });

  app.put('/allocation-targets', async (request: FastifyRequest, reply: FastifyReply) => {
    const { targets } = request.body as { targets: { category: string; targetRatio: number }[] };
    const ops = targets.map((t) =>
      prisma.allocationTarget.upsert({
        where: { userId_category: { userId, category: t.category } },
        update: { targetRatio: t.targetRatio },
        create: { userId, category: t.category, targetRatio: t.targetRatio },
      })
    );
    await prisma.$transaction(ops);
    const updated = await prisma.allocationTarget.findMany({ where: { userId } });
    return reply.send(updated);
  });

  app.get('/allocation-analysis', async (_request: FastifyRequest, reply: FastifyReply) => {
    const assets = await prisma.asset.findMany({ where: { userId } });
    const targets = await prisma.allocationTarget.findMany({ where: { userId } });
    const totalValue = assets.reduce((sum, a) => sum + a.totalValue, 0);

    const currentByCategory: Record<string, number> = {};
    for (const a of assets) {
      currentByCategory[a.category] = (currentByCategory[a.category] || 0) + a.totalValue;
    }

    const analysis: AllocationAnalysis[] = targets.map((t) => {
      const currentValue = currentByCategory[t.category] || 0;
      const currentRatio = totalValue > 0 ? (currentValue / totalValue) * 100 : 0;
      const deviation = currentRatio - t.targetRatio;
      const targetValue = totalValue * (t.targetRatio / 100);
      const rebalanceAmount = targetValue - currentValue;

      return {
        category: t.category as AllocationAnalysis['category'],
        currentValue: Math.round(currentValue * 100) / 100,
        currentRatio: Math.round(currentRatio * 100) / 100,
        targetRatio: t.targetRatio,
        deviation: Math.round(deviation * 100) / 100,
        rebalanceAmount: Math.round(rebalanceAmount * 100) / 100,
      };
    });

    return reply.send(analysis);
  });
}
```

```typescript
const BASE = '/api';

async function request<T>(url: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE}${url}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  });
  if (!res.ok) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body.error || `Request failed: ${res.status}`);
  }
  if (res.status === 204) return undefined as T;
  return res.json();
}

export const api = {
  assets: {
    list: (category?: string) => request(`/assets${category ? `?category=${category}` : ''}`),
    create: (data: unknown) => request('/assets', { method: 'POST', body: JSON.stringify(data) }),
    update: (id: string, data: unknown) => request(`/assets/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
    delete: (id: string) => request(`/assets/${id}`, { method: 'DELETE' }),
  },
  liabilities: {
    list: () => request('/liabilities'),
    create: (data: unknown) => request('/liabilities', { method: 'POST', body: JSON.stringify(data) }),
    update: (id: string, data: unknown) => request(`/liabilities/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
    delete: (id: string) => request(`/liabilities/${id}`, { method: 'DELETE' }),
  },
  netWorth: {
    get: () => request('/net-worth'),
    history: () => request('/net-worth/history'),
  },
  fireProfile: {
    get: () => request('/fire-profile'),
    update: (data: unknown) => request('/fire-profile', { method: 'PUT', body: JSON.stringify(data) }),
    progress: () => request('/fire-progress'),
  },
  allocation: {
    targets: () => request('/allocation-targets'),
    updateTargets: (data: unknown) => request('/allocation-targets', { method: 'PUT', body: JSON.stringify(data) }),
    analysis: () => request('/allocation-analysis'),
  },
  snapshots: {
    create: () => request('/snapshots', { method: 'POST' }),
    list: () => request('/snapshots'),
  },
};
```

import { NavLink } from 'react-router-dom';
import { LayoutDashboard, Target, PieChart, Moon, Sun } from 'lucide-react';
import { useState, useEffect } from 'react';

const links = [
  { to: '/', icon: LayoutDashboard, label: '资产总览' },
  { to: '/fire', icon: Target, label: 'FIRE 追踪' },
  { to: '/allocation', icon: PieChart, label: '配置分析' },
];

export default function NavBar() {
  const [dark, setDark] = useState(() => document.documentElement.classList.contains('dark'));

  useEffect(() => {
    document.documentElement.classList.toggle('dark', dark);
  }, [dark]);

  return (
    <nav className="h-14 border-b flex items-center justify-between px-6 bg-card">
      <div className="flex items-center gap-1">
        <span className="text-xl font-bold text-primary mr-6">free</span>
        {links.map(({ to, icon: Icon, label }) => (
          <NavLink
            key={to}
            to={to}
            end={to === '/'}
            className={({ isActive }) =>
              `flex items-center gap-2 px-3 py-2 rounded-md text-sm transition-colors ${
                isActive
                  ? 'bg-primary/10 text-primary font-medium'
                  : 'text-muted-foreground hover:text-foreground hover:bg-secondary'
              }`
            }
          >
            <Icon className="h-4 w-4" />
            {label}
          </NavLink>
        ))}
      </div>
      <button
        onClick={() => setDark(!dark)}
        className="p-2 rounded-md hover:bg-secondary text-muted-foreground"
      >
        {dark ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
      </button>
    </nav>
  );
}
```

import { useState, useEffect, useCallback } from 'react';
import { api } from '../lib/api';
import type { Asset, CreateAssetInput, UpdateAssetInput } from '../types';

export function useAssets() {
  const [assets, setAssets] = useState<Asset[]>([]);
  const [loading, setLoading] = useState(true);

  const fetch = useCallback(async () => {
    setLoading(true);
    const data = await api.assets.list();
    setAssets(data);
    setLoading(false);
  }, []);

  useEffect(() => { fetch(); }, [fetch]);

  const create = async (input: CreateAssetInput) => { await api.assets.create(input); await fetch(); };
  const update = async (id: string, input: UpdateAssetInput) => { await api.assets.update(id, input); await fetch(); };
  const remove = async (id: string) => { await api.assets.delete(id); await fetch(); };

  return { assets, loading, create, update, remove, refetch: fetch };
}
```
